C++ Quick Reference Cheat Sheet

Core Syntax, Pointers, Memory & STL Containers

1. Essential Syntax & Types

// Auto Type Deduction
auto x = 42;          // int
auto pi = 3.14159;    // double

// References vs Pointers
int val = 100;
int& ref = val;       // alias to val
int* ptr = &val;      // pointer to address

2. Modern Memory & Smart Pointers

#include <memory>

// Unique Ownership (Zero Overhead)
auto uPtr = std::make_unique<int>(50);

// Shared Ownership (Ref Counted)
auto sPtr = std::make_shared<std::string>("Hi");
// No delete required! (Automatic RAII)

3. STL Containers

std::vector<int> v = {1, 2, 3};
v.push_back(4);       // O(1) amortized

std::unordered_map<string, int> mp;
mp["Alice"] = 95;     // O(1) avg lookup

std::sort(v.begin(), v.end()); // O(n log n)

4. Classes & OOP

class Shape {
public:
    virtual void draw() const = 0; // Pure virtual
    virtual ~Shape() = default;    // Virtual dtor
};

class Circle : public Shape {
    void draw() const override { /*...*/ }
};
